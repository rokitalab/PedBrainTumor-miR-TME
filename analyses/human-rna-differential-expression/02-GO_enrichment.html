Run GO enrichment for upregulated genes

all_genes <- de_mrna_df$gene_name[!is.na(de_mrna_df$gene_name)]

is_upregulated <- ifelse(all_genes %in% upregulated_genes,
                         1, 0)
names(is_upregulated) <- all_genes

# Create topGO object, using biological process (BP) ontology terms
GO_data_up <- new("topGOdata", ontology = "BP", allGenes = is_upregulated, 
                                                 geneSel = function(x) x == 1, 
                                                 annot = annFUN.org, mapping = "org.Hs.eg.db", ID = "symbol")

# write resutls to go analysis list
GO_table_up <- GenTable(GO_data_up, classicFisher = GO_results_up, topNodes = length(usedGO(GO_data_up)),
                                                     numChar = 100000)

sig_table_up <- GO_table_up %>%
  dplyr::mutate(classicFisher = as.numeric(classicFisher)) %>%
  dplyr::mutate(fdr = p.adjust(classicFisher, method = "BH"),
                 gene_ratio = Significant/Annotated) %>%
  dplyr::arrange(desc(gene_ratio)) %>%
  dplyr::filter(fdr < 0.05)
