This script summarizes immune deconvolution results as follows:

  • Compares immune cell fractions and scores across tumor histologies

Loop through histologies and plot cell fractions in tumor vs. normal

groups <- c("DMG", "EPN", "LGG", "MB")

for (hist in groups){
  
  hist_plot_dir <- file.path(plot_dir, hist)
  
  if (!dir.exists(hist_plot_dir)){
    
    dir.create(hist_plot_dir)
    
  }
  
  # subset for histology
  quantiseq_group <- quantiseq_res %>%
    dplyr::filter(histology == hist | sample_type == "Normal",
                cell_type != "uncharacterized cell")

# plot tumor vs adj and healthy normal
  quantiseq_group %>%
    dplyr::mutate(group = fct_relevel(group,
                                      c("Healthy N",
                                        "Adjacent N",
                                        "Tumor"))) %>%
    dplyr::arrange(group) %>%
    
    ggplot(aes(x = factor(group), y = fraction)) + 
    geom_jitter(aes(fill = group), shape = 21, 
                width = 0.2, size = 2.5, alpha = 0.85,
                show.legend = FALSE) +
    geom_boxplot(alpha = 0.05, outlier.shape = NA) +
    stat_compare_means(method = "wilcox",
                       comparisons = list(c("Tumor", "Healthy N"),
                                          c("Tumor", "Adjacent N")),
                       method.args = list(alternative = "two.sided")) +
    labs(x = NULL, y = "Cell fraction") +
    facet_wrap(~cell_type, nrow = 2, scales = "free_y",
               labeller = labeller(cell_type = label_wrap_gen(18))) +
    scale_y_continuous(expand = expansion(mult = c(0.05, 0.2))) +
    theme_Publication() +
    theme(axis.text.x = element_text(angle = 45, hjust = 1))
    
  ggsave(file.path(hist_plot_dir,
                   glue::glue("{hist}-cell-proportions-tumor-v-adj-healthyN-quantiseq.pdf")),
         width = 15, height = 6)
    
  # plot paired tumor-adj samples
  if (hist != "LGG"){
  
    quantiseq_group %>%
      dplyr::filter(!is.na(histology),
                    sample_id %in% paired_samples) %>%
      dplyr::arrange(cell_type, sample_id, group) %>%
      
      ggplot(aes(x = group, y = fraction)) + 
      geom_jitter(aes(fill = group), shape = 21, 
                  width = 0.2, size = 2.5, alpha = 0.85,
                  show.legend = FALSE) +
      geom_boxplot(alpha = 0.05, outlier.shape = NA) +
      stat_compare_means(method = "wilcox",
                         comparisons = list(c("Tumor", "Adjacent N")),
                         method.args = list(alternative = "two.sided")) +
      labs(x = NULL, y = "Cell fraction") +
      facet_wrap(~cell_type, nrow = 2, scales = "free_y",
                 labeller = labeller(cell_type = label_wrap_gen(18))) +
      scale_y_continuous(expand = expansion(mult = c(0.05, 0.2))) +
      theme_Publication()
    
    ggsave(file.path(hist_plot_dir,
                 glue::glue("{hist}-cell-proportions-tumor-v-adjN-quantiseq.pdf")),
         width = 12, height = 5)
      
  }
  
}
