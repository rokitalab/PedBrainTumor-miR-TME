Run differential expression analyses

We will generate DESeqDataSet objects for each timepoint and perform the following comparisions:

  • B7H3 versus untreated
  • STOP versus untreated
# create empty results list
res <- list()


for (time in time_points){
  
  dds <- DESeqDataSetFromMatrix(countData = gene_cts[, metadata_df$time == time],
                                colData = metadata_df[metadata_df$time == time, ],
                                design = ~ treatment)
  
  dds <- DESeq(dds)
  
  # pull B7H3 versus untreated contrast and generate data frame
  res_B7H3_vs_Untreated <- results(dds, 
                                     contrast = c("treatment", "B7H3", "Untreated"))
  
  res_B7H3_vs_Untreated <- data.frame(res_B7H3_vs_Untreated) %>%
    rownames_to_column("gene_symbol")
  
  
  # generate volcano plot
  volc_plot <- EnhancedVolcano(res_B7H3_vs_Untreated,
                           lab = res_B7H3_vs_Untreated$gene_symbol, 
                           x = "log2FoldChange",
                           y = "pvalue",
                           title = glue::glue("B7H3 CAR-T versus Untreated, {time}"),
                           subtitle = NULL,
                           caption = NULL,
                           pCutoff = max(res_B7H3_vs_Untreated$pvalue[res_B7H3_vs_Untreated$padj < 0.05], na.rm = TRUE),
                           FCcutoff = 2,
                           pointSize = 4,
                           labSize = 4,
                           typeConnectors = "closed",
                           drawConnectors = TRUE,
                           widthConnectors = 1,
                           colConnectors = 'black',
                           max.overlaps = 30)

  ggsave(file.path(plot_dir,
                glue::glue("murine-rna-deseq2-b7h3-versus-untreated-{time}.pdf")),
         volc_plot, width = 8, height = 6)
  
  
  # pull STOP versus untreated contrast and generate data frame
  res_STOP_vs_untreated <- results(dds, 
                                  contrast = c("treatment", "STOP", "Untreated"))
  
  res_STOP_vs_untreated <- data.frame(res_STOP_vs_untreated) %>%
    rownames_to_column("gene_symbol")
  
  # join two contrast dfs by gene_symbol, adding suffixes to column names to indicate comparisons
  res[[time]] <- res_B7H3_vs_Untreated %>%
    left_join(res_STOP_vs_untreated,
              suffix = c("_b7h3_vs_untr",
                         "_stop_vs_untr"),
              by = "gene_symbol") %>%
    # define B7H3-upregulated and -downregulated genes as those DE in B7H3 versus untreated ONLY, and NOT DE in STOP versus untreated comparison 
    dplyr::mutate(status = case_when(
      padj_b7h3_vs_untr < 0.05 & log2FoldChange_b7h3_vs_untr > 0 & padj_stop_vs_untr > 0.05 ~ "B7H3 upregulated",
      padj_b7h3_vs_untr < 0.05 & log2FoldChange_b7h3_vs_untr < 0 & padj_stop_vs_untr > 0.05 ~ "B7H3 downregulated",
      TRUE ~ "Not DE"
    ))
}

Merge DE results by time point, and classify genes based on differential expression patterns over time

# join timepoint dfs
res_all <- res[["Day14"]] %>%
  left_join(res[["Day21"]],
            by = "gene_symbol",
            suffix = c("_Day14", "_Day21")) %>%
  # group genes by differential expression patterns
  dplyr::mutate(expr_pattern = case_when(
    status_Day14 == "B7H3 upregulated" & status_Day21 == "Not DE" ~ "B7H3 up, Day 14",
    status_Day14 == "Not DE" & status_Day21 == "B7H3 upregulated" ~ "B7H3 up, Day 21",
    status_Day14 == "B7H3 downregulated" & status_Day21 == "Not DE" ~ "B7H3 down, Day 14",
    status_Day14 == "Not DE" & status_Day21 == "B7H3 downregulated" ~ "B7H3 down, Day 21",
    status_Day14 == "B7H3 upregulated" & status_Day21 == "B7H3 upregulated" ~ "B7H3 up, multiple",
    status_Day14 == "B7H3 downregulated" & status_Day21 == "B7H3 downregulated" ~ "B7H3 down, multiple",
    status_Day14 != "Not DE" | status_Day21 != "Not DE" ~ "Other DE",
    TRUE ~ "Not DE"
  ))
