Load sample metadata df

metadata_df <- read_delim(file.path(data_dir, "miRNA-sample-metadata.txt")) %>%
  mutate(histology = if_else(histology == "N/A", "Normal", histology)) %>%
  mutate(patient_id = str_remove(sample_id, "-[TN]$")) %>%
  filter(sample_id %in% names(mirna_cts)) %>%
  arrange(match(sample_id, names(mirna_cts))) %>%
  mutate(batch = if_else(row_number() <= 5, "batch1", "batch2")) %>%
  mutate(condition = case_when(
    sample_type %in% c("Tumor", "Normal") ~ histology,
    sample_type == "Normal-Tumor Adjacent" ~ paste0(histology, "-adjacent Normal"),
    TRUE ~ NA_character_
  )) %>%
  mutate(combined_adjacent = case_when(
    sample_type %in% c("Tumor", "Normal") ~ histology,
    sample_type == "Normal-Tumor Adjacent" ~ "Normal-Tumor Adjacent",
    TRUE ~ NA_character_
  ))

Run UMAP

set.seed(42)

# Default UMAP config
umap_config <- umap.defaults
umap_config$n_neighbors <- 15
umap_config$min_dist <- 0.3

# UMAP
umap_result <- umap(t(assay(vsd)), config = umap_config)

umap_df <- as.data.frame(umap_result$layout)
colnames(umap_df) <- c("UMAP1", "UMAP2")
umap_df$sample_id <- rownames(umap_df)

# Join with metadata
umap_df <- left_join(umap_df, metadata_df, by = "sample_id")
