# Read and tag each file
read_de_file <- function(path, subtype, comparison) {
  df <- read_csv(path, show_col_types = FALSE)
  df %>%
    mutate(
      subtype = subtype,
      comparison = comparison
    )
}

# Filter up/down regulated
get_up_down_miRNAs <- function(df) {
  upregulated <- df %>% filter(log2FoldChange > 0) %>% pull(miRNA) %>% unique()
  downregulated <- df %>% filter(log2FoldChange < 0) %>% pull(miRNA) %>% unique()
  list(up = upregulated, down = downregulated)
}

# Extract unique/shared miRNAs
extract_venn_table <- function(set1, set2, label1, label2) {
  shared <- intersect(set1, set2)
  unique1 <- setdiff(set1, set2)
  unique2 <- setdiff(set2, set1)
  tibble(
    Category = c(rep(label1, length(unique1)),
                 rep(label2, length(unique2)),
                 rep("Shared", length(shared))),
    miRNA = c(unique1, unique2, shared)
  )
}

# Convert named list -> membership data.frame for UpSet
as_membership_df <- function(named_list) {
  all_items <- sort(unique(unlist(named_list)))
  mat <- sapply(named_list, function(v) all_items %in% v)
  df <- data.frame(miRNA = all_items, mat, check.names = FALSE)
  return(df)
}

# Build one list per subtype for up- and down-regulated
to_named_list <- function(df, dir_label) {
  tmp <- df %>%
    filter(direction == dir_label) %>%
    group_by(subtype) %>%
    summarise(miRNAs = list(unique(miRNA)), .groups = "drop")
  setNames(tmp$miRNAs, tmp$subtype)
}

Venn diagrams per subtype

subtypes_main <- c("DIPG or DMG", "MB", "EPN")
subtypes_full <- c("DIPG or DMG", "MB", "EPN", "LGG")

for (tumor in subtypes_main) {
  # Grab the two tables for this subtype
  paired_adj <- de_tbl %>% filter(subtype == tumor, comparison == "tumor_vs_adjacent")
  healthy_norm <- de_tbl %>% filter(subtype == tumor, comparison == "tumor_vs_healthynormal")

  # Up/down lists
  paired_sets   <- get_up_down_miRNAs(paired_adj)
  healthy_sets  <- get_up_down_miRNAs(healthy_norm)

  # ---------- Upregulated ----------
  venn_up <- list(
    `Paired T vs AN` = paired_sets$up,
    `T vs HN`        = healthy_sets$up
  )

  p_up <- ggvenn(
    venn_up,
    fill_color = c("#E69F00", "#56B4E9"),
    show_percentage = TRUE,
    set_name_size = 5
  ) +
    ggtitle(paste("Upregulated miRNAs –", tumor)) +
    theme(
      plot.title = element_text(hjust = 0.5, vjust = 5, size = 16, margin = margin(b = 5)),
      plot.title.position = "plot",
      plot.margin = margin(5, 5, 5, 5)
    )

  ggsave(
    file.path(plot_dir, paste0("venn_up_paired_vs_healthy_", tumor, ".pdf")),
    plot = p_up, width = 6, height = 5
  )

  venn_table_up <- extract_venn_table(paired_sets$up, healthy_sets$up, "Paired T vs AN", "T vs HN")
  write_csv(venn_table_up, file.path(results_dir, paste0("venn_table_up_paired_vs_healthy_", tumor, ".csv")))

  # ---------- Downregulated ----------
  venn_down <- list(
    `Paired T vs AN` = paired_sets$down,
    `T vs HN`        = healthy_sets$down
  )

  p_down <- ggvenn(
    venn_down,
    fill_color = c("#E69F00", "#56B4E9"),
    show_percentage = TRUE,
    set_name_size = 5
  ) +
    ggtitle(paste("Downregulated miRNAs –", tumor)) +
    theme(
      plot.title = element_text(hjust = 0.5, vjust = 5, size = 16, margin = margin(b = 5)),
      plot.title.position = "plot",
      plot.margin = margin(5, 5, 5, 5)
    )

  ggsave(
    file.path(plot_dir, paste0("venn_down_paired_vs_healthy_", tumor, ".pdf")),
    plot = p_down, width = 6, height = 5
  )

  venn_table_down <- extract_venn_table(paired_sets$down, healthy_sets$down, "Paired T vs AN", "T vs HN")
  write_csv(venn_table_down, file.path(results_dir, paste0("venn_table_down_paired_vs_healthy_", tumor, ".csv")))

  # Combined list for this subtype
  venn_table_up$direction   <- "up"
  venn_table_down$direction <- "down"
  combined_sig_de <- bind_rows(venn_table_up, venn_table_down)
  write_csv(combined_sig_de, file.path(results_dir, paste0(tumor, "_sig_DE_miRNA_list.csv")))
}
