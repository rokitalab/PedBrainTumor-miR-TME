cluster_summary_df <- tibble(
                             # repeat miR clusters to test for enrichment within each mouse DE group
                             hsa_dmg_cluster = rep(unique(dmg_hsa_mir_df$row_cluster), each = 5),
                             de_mir_group = rep(names(de_mir_list), 8),
                             total_n = NA_integer_,
                             n_mmu_orthologs = NA_integer_,
                             n_expr_orthologs = NA_integer_,
                             n_expr_group = NA_integer_,
                             n_mmu_mir = sum(!grepl("Novel", mmu_de_df$mirna_id)),
                             OR = NA_integer_,
                             OR_p = NA_integer_,
                             OR_CI_lower = NA_integer_,
                             OR_CI_upper = NA_integer_)

Loop through rows of cluster_summary_df to 1) identify number of total and expressed mouse orthologs per human DMG miR cluster 2) Assess enrichment of orthologs for miRNAs of different B7H3-associated expression patterns

for (i in 1:nrow(cluster_summary_df)){
  
  # get cluster
  clust <- cluster_summary_df$hsa_dmg_cluster[i]
  
  # get miRNAs in cluster
  cluster_mirs <- dmg_hsa_mir_df %>%
    dplyr::filter(row_cluster == clust) %>%
    # remove hsa prefix from miRNA ID
    dplyr::mutate(miRNA = str_remove(miRNA, "hsa-")) %>%
    pull(miRNA)

  # after removing species prefix, orthologous miRNAs will share the same name between species
  # determine how many hsa DMG cluster miRNAs have orthologs in mice:
  cluster_mirs_w_ortholog <- mmu_mirna_ct_df %>%
    dplyr::filter(mirna_id %in% cluster_mirs) %>%
    pull(mirna_id)

  # get N expressed orthologs from DE df 
  expr_orthologs <- mmu_de_df %>% 
    dplyr::filter(mirna_id %in% cluster_mirs) %>%
    pull(mirna_id)
  
  # append stats to summary df
  cluster_summary_df$total_n[i] <- length(cluster_mirs)
  cluster_summary_df$n_mmu_orthologs[i] <- length(cluster_mirs_w_ortholog)
  cluster_summary_df$n_expr_orthologs[i] <- length(expr_orthologs)
  
  # define current DE group and miRs in group
  de_group <- cluster_summary_df$de_mir_group[i]
  
  de_mirs <- de_mir_list[[de_group]]
  
  # determine number of orthologs in cluster and expression group
  n_ortho_de <- length(intersect(expr_orthologs,
                                 de_mirs))
  
  cluster_summary_df$n_expr_group[i] <- n_ortho_de
  
  # run fisher's exact test to assess for enrichment of orthologs in DE group
  ftest <- fisher.test(matrix(c(n_ortho_de,
                                length(expr_orthologs) - n_ortho_de,
                                length(de_mirs) - n_ortho_de,
                                cluster_summary_df$n_mmu_mir[i] - n_ortho_de - (length(expr_orthologs) - n_ortho_de) - (length(de_mirs) - n_ortho_de)),
                              nrow = 2))
  
  # append FET stats to summary df 
  cluster_summary_df$OR[i] <- ftest$estimate
  cluster_summary_df$OR_p[i] <- ftest$p.value
  cluster_summary_df$OR_CI_lower[i] <- ftest$conf.int[1]
  cluster_summary_df$OR_CI_upper[i] <- ftest$conf.int[2]
 
}

Generate df to plot % orthologs identified by cluster

ortho_plot_df <- cluster_summary_df %>%
  # retain one row per cluster
  distinct(hsa_dmg_cluster, .keep_all = TRUE) %>%
  dplyr::select(hsa_dmg_cluster,
                total_n,
                n_mmu_orthologs,
                n_expr_orthologs) %>%
  dplyr::rename(`Expressed ortholog` = n_expr_orthologs) %>%
  dplyr::mutate(`Non-expressed ortholog` = n_mmu_orthologs - `Expressed ortholog`,
                `No ortholog` = total_n - n_mmu_orthologs) %>%
  dplyr::select(-n_mmu_orthologs, -total_n) %>%
  pivot_longer(-hsa_dmg_cluster,
               names_to = "category",
               values_to = "n") %>%
  dplyr::mutate(category = fct_relevel(category,
                                       c("Expressed ortholog",
                                         "Non-expressed ortholog",
                                         "No ortholog"))) %>%
  dplyr::mutate(hsa_dmg_cluster = as.factor(hsa_dmg_cluster)) %>%
  dplyr::mutate(hsa_dmg_cluster = fct_relevel(hsa_dmg_cluster,
                                              rev(unique(as.character(hsa_dmg_cluster)))))

  dplyr::mutate(p_text = case_when(
    OR_p < 0.01 ~ "**",
    OR_p < 0.05 ~ "*",
    TRUE ~ ""
  ))

Plot enrichment of miRNAs in B7H3-associated expression groups in mouse by cluster

enr_plot <- cluster_summary_df %>% 
  # remove clusters with <2 orthologous miRNAs
  dplyr::filter(!hsa_dmg_cluster %in% c(2,7)) %>%
  dplyr::mutate(OR = case_when(
    is.infinite(OR) ~ 1000,
    TRUE ~ OR
  )) %>%
  dplyr::mutate(OR_CI_upper = case_when(
    is.infinite(OR_CI_upper) ~ 100,
    TRUE ~ OR_CI_upper
  )) %>%
  ggplot(aes(x = log10(OR), y = factor(de_mir_group),
             color = factor(de_mir_group),
             label = p_text)) +
  geom_point(size = 3, show.legend = FALSE) + 
  # add log10-transformed confidence intervals
  geom_errorbar(aes(xmin = log10(OR_CI_lower), xmax = log10(OR_CI_upper),
                    color = factor(de_mir_group)), width = 0.2, 
                show.legend = FALSE) +
  geom_text(x = 1.5, hjust = 0, size = 6, fontface = 2,
            show.legend = FALSE) +
  labs(x = "log10-Odds Ratio\n(95% CI)", y = "B7H3 DE miRNA class") + 
  # Add OR threshold line
  geom_vline(xintercept = 0, linetype = "dashed") +
  # facet wrap by variable, place strips to left
  facet_wrap(~hsa_dmg_cluster, scale = "fixed",
             nrow = 8,
             strip.position = "left") +
  expand_limits(y=0) +
  scale_color_manual(values = c("#3C5488FF", "#E64B35FF",
                                "#4DBBD5FF", "#F39B7FFF",
                                "#00A087FF")) +
  theme_Publication()
